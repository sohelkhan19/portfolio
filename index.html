<?php
namespace SohelPlugin\Includes;

class Ajax_Handler {

    public function __construct() {
        add_action('wp_ajax_filter_business_portfolio', [$this, 'handle_filter']);
        add_action('wp_ajax_nopriv_filter_business_portfolio', [$this, 'handle_filter']);
    }

    public function handle_filter() {
        // 1. Validate Cryptographic Nonce
        check_ajax_referer('business_portfolio_nonce', 'security');

        // 2. Sanitize Request Parameters
        $category = !empty($_POST['category']) ? sanitize_key($_POST['category']) : 'all';

        // 3. Performance Query without SQL count calculation
        $args = [
            'post_type'      => 'business_portfolio',
            'post_status'    => 'publish',
            'posts_per_page' => 12,
            'no_found_rows'  => true,
        ];

        $query = new \WP_Query($args);
        $items = [];
        while ($query->have_posts()) {
            $query->the_post();
            $items[] = [
                'id'    => get_the_ID(),
                'title' => esc_html(get_the_title()),
            ];
        }
        wp_reset_postdata();
        wp_send_json_success(['items' => $items]);
    }
}

app/Http/Controllers/Api/V1/ProjectController.php 
<?php
namespace App\Http\Controllers\Api\V1;

use App\Http\Controllers\Controller;
use App\Http\Requests\StoreProjectRequest;
use App\Http\Resources\ProjectResource;
use App\Models\Project;
use App\Jobs\ProcessProjectExport;
use Illuminate\Http\JsonResponse;
use Illuminate\Support\Facades\DB;

class ProjectController extends Controller {

    public function index(): JsonResponse {
        // Eager loading relations with specific column selections
        $projects = Project::with(['category:id,name,slug', 'team:id,name'])
            ->where('organization_id', auth()->user()->organization_id)
            ->latest()
            ->paginate(15);

        return response()->json([
            'status' => 'success',
            'data'   => ProjectResource::collection($projects),
        ]);
    }

    public function store(StoreProjectRequest $request): JsonResponse {
        $project = DB::transaction(function () use ($request) {
            $created = Project::create($request->validated() + [
                'organization_id' => auth()->user()->organization_id,
            ]);

            // Dispatch async queue worker via Redis
            ProcessProjectExport::dispatch($created)->onQueue('reports');
            return $created;
        });

        return response()->json([
            'status'  => 'success',
            'message' => 'Project created and queued for processing.',
            'data'    => new ProjectResource($project->load('category'))
        ], 201);
    }
}
